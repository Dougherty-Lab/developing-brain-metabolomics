Overview

ORA + GSEA on the genes from gene–metabolite associations in the log2_na run (metabolite_gene_associations.qmd). The companion script metabolite_pathway_analysis.qmd does the same for metabolites. See gene_pathway_analysis_dfbetas.qmd for the DFBETAS-stable robust subset.

  • ORA: Fisher’s exact test; foreground = genes with ≥1 global-FDR association; background = all genes tested in the limma-voom loop.
  • GSEA: fgsea ranked by −log10(min raw P-value per gene) across all metabolite × cell type pairs. Unsigned — NES > 0 means the gene set’s members are enriched for metabolite associations.

Three analyses:

  1. All cell types (union) — genes significant in ≥1 cell type
  2. Cell-type-specific — ORA + GSEA per cell type
  3. Glia vs. neurons — class-specific gene sets and shared pathway comparison

Gene sets from msigdbr (MSigDB v2023.2): GO:BP, GO:MF, GO:CC, Hallmark, KEGG, Reactome. Note: msigdbr must be added to the Singularity container rebuild (developing-brain-metabolomics_1.0.4.sif → next version).

# Anchor output paths to the repo root (found via the .git marker) instead of
# a cwd-relative "../.." — re-running chunks out of order in an interactive
# session can leave the R working directory pointed somewhere unexpected
# (e.g. a subfolder), which otherwise silently writes results under src/.
find_project_root <- function(marker = ".git") {
  d <- normalizePath(getwd())
  repeat {
    if (dir.exists(file.path(d, marker))) return(d)
    parent <- dirname(d)
    if (parent == d) stop("Could not locate project root (no ", marker, " found above ", getwd(), ")")
    d <- parent
  }
}

# ---- ORA (Fisher's exact, one-sided greater) ---------------------------------
run_gene_ora <- function(foreground, background, gene_set_list,
                         gene_set_source_lookup, analysis_label) {
  cat("\n===", analysis_label, "===\n")
  cat("Foreground genes:", length(foreground),
      "| Background genes:", length(background), "\n")

  if (length(foreground) < 1) { cat("Empty foreground — skipping.\n"); return(NULL) }

  bg_only <- background[!background %in% foreground]

  results <- map_dfr(names(gene_set_list), function(gs) {
    members <- gene_set_list[[gs]]
    a <- sum(foreground %in% members)
    b <- length(foreground) - a
    c <- sum(bg_only %in% members)
    d <- length(bg_only) - c
    ft <- fisher.test(matrix(c(a, b, c, d), nrow = 2), alternative = "greater")
    tibble(
      gene_set   = gs,
      n_hits     = a,
      set_size   = a + c,
      odds_ratio = as.numeric(ft$estimate),
      pval       = ft$p.value,
      hit_genes  = paste(foreground[foreground %in% members], collapse = ";")
    )
  }) |>
    filter(n_hits > 0) |>
    mutate(padj = p.adjust(pval, method = "BH")) |>
    left_join(gene_set_source_lookup, by = c("gene_set" = "gs_label")) |>
    arrange(padj, pval)

  cat("Sets with \u22651 hit:", nrow(results),
      "| p < 0.05:", sum(results$pval  < 0.05, na.rm = TRUE),
      "| FDR < 0.05:", sum(results$padj < 0.05, na.rm = TRUE), "\n")
  results
}

# ---- GSEA (fgsea on −log10 min P per gene) -----------------------------------
run_gene_gsea <- function(ranking_df, background_genes, gene_set_list,
                          analysis_label, nperm = 10000, seed = 42) {
  cat("\n===", analysis_label, "===\n")

  ranked <- tibble(gene = background_genes) |>
    left_join(ranking_df |> dplyr::select(gene, min_P), by = "gene") |>
    mutate(score = -log10(pmax(replace_na(min_P, 1), 1e-300))) |>
    arrange(desc(score)) |>
    dplyr::select(gene, score) |>
    deframe()

  cat("Genes in ranked list:", length(ranked), "\n")

  gs_filt <- lapply(gene_set_list, function(x) intersect(x, names(ranked)))
  gs_filt <- gs_filt[lengths(gs_filt) >= GSEA_MIN_SIZE]
  cat("Gene sets after intersection:", length(gs_filt), "\n")

  if (length(gs_filt) == 0) {
    cat("No gene sets with sufficient coverage — skipping.\n")
    return(NULL)
  }

  set.seed(seed)
  results <- fgsea(
    pathways    = gs_filt,
    stats       = ranked,
    nPermSimple = nperm,
    minSize     = GSEA_MIN_SIZE,
    maxSize     = GSEA_MAX_SIZE
  ) |>
    as_tibble() |>
    arrange(padj, pval) |>
    mutate(analysis = analysis_label)

  cat("Sets tested:", nrow(results),
      "| p < 0.05:", sum(results$pval  < 0.05, na.rm = TRUE),
      "| FDR < 0.05:", sum(results$padj < 0.05, na.rm = TRUE), "\n")
  results
}

plot_ora_dotplot <- function(results, analysis_label, top_n = 20,
                             collections = NULL) {
  if (is.null(results) || nrow(results) == 0) return(NULL)

  if (!is.null(collections)) results <- filter(results, collection %in% collections)

  plot_data <- results |>
    filter(pval < 0.05) |>
    head(top_n) |>
    mutate(
      stars     = case_when(padj < 0.01 ~ "**", padj < 0.05 ~ "*", TRUE ~ ""),
      gs_base   = str_remove(gene_set, " \\([^)]+\\)$"),
      gs_short  = str_trunc(gs_base, 45),
      gs_label  = paste0(gs_short, " (", collection, ")",
                         if_else(stars != "", paste0(" ", stars), ""))

  if (nrow(plot_data) == 0) { message("No nominal hits for: ", analysis_label); return(NULL) }

  ggplot(plot_data,
         aes(x = log2(odds_ratio),
             y = reorder(gs_label, log2(odds_ratio)),
             color = collection, shape = collection, size = set_size)) +
    geom_point(alpha = 0.85) +
    geom_vline(xintercept = 0, linetype = "dashed", color = "grey50") +
    scale_color_manual(name = "Collection", values = COLL_COLORS) +

# ---- Table helper ------------------------------------------------------------
display_results_table <- function(results, caption, top_n = 20,
                                  type = c("ora", "gsea"),
                                  gene_set_source_lookup = NULL) {
  type <- match.arg(type)
  if (is.null(results) || nrow(results) == 0) { cat("No results.\n"); return(invisible(NULL)) }

  if (type == "ora") {
    results |>
      head(top_n) |>
      mutate(odds_ratio = round(odds_ratio, 2),
             pval = signif(pval, 3), padj = signif(padj, 3),
             hit_genes = str_trunc(hit_genes, 60)) |>
      dplyr::select(gene_set, collection, n_hits, set_size,
                    odds_ratio, pval, padj, hit_genes) |>
      rename(`Gene set` = gene_set, Collection = collection,
             Hits = n_hits, `Set size` = set_size,
             OR = odds_ratio, `p-value` = pval, FDR = padj,
             `Hit genes` = hit_genes) |>
      kable(caption = caption) |>
      kable_styling(bootstrap_options = c("striped", "condensed"),
                    full_width = FALSE, font_size = 11) |>
      print()
  } else {
    df <- results
    if (!is.null(gene_set_source_lookup))
      df <- df |> left_join(gene_set_source_lookup, by = c("pathway" = "gs_label"))
    df |>
      head(top_n) |>
      mutate(pval = signif(pval, 3), padj = signif(padj, 3),
             NES  = round(NES, 3),
             leadingEdge = map_chr(leadingEdge, ~ paste(head(.x, 5), collapse = ", "))) |>
      dplyr::select(pathway, collection, pval, padj, NES, size, leadingEdge) |>
      rename(`Gene set` = pathway, Collection = collection,
             `p-value` = pval, FDR = padj, `Set size` = size,
             `Leading edge (top 5)` = leadingEdge) |>
      kable(caption = caption) |>
      kable_styling(bootstrap_options = c("striped", "condensed"),
                    full_width = FALSE, font_size = 11) |>
      print()
  }
}

flatten_leading_edge <- function(df) {
  if (!is.null(df) && "leadingEdge" %in% colnames(df))
    df |> mutate(leadingEdge = map_chr(leadingEdge, ~ paste(.x, collapse = ";")))
  else df
}
