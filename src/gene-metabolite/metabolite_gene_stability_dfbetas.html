Metabolite–Gene Stability Filter — DFBETAS (log2_na run)

Sneha Chaturvedi

2026-08-12

Purpose

Second-gate filter producing the most robust gene–metabolite associations. Input is the global-FDR hit set from metabolite_gene_associations.qmd (the log2_na run: no imputation, IQR outlier trimming, z-scored predictor). Because log2_na carries no imputed values, the imputation-sensitivity gate is not needed and this script reads global-FDR hits directly.

Associations are kept only if no single sample has outsized influence on the metabolite slope. Uses DFBETAS — the standardized single-deletion change in the coefficient, in units of its own SE — with the standard cutoff |DFBETAS| > 2/sqrt(n) (Belsley, Kuh & Welsch 1980). This answers “does dropping one point matter”, not merely whether a high-leverage point exists, so a far-x sample consistent with the trend is kept.

Pairs that survive the DFBETAS gate are written as metabolite_gene_hits_robust.csv and represent the highest-confidence set for manuscript-level reporting.

add_plot_label <- function(df, name_col = "Name", n = LABEL_CHARS) {
  short <- ifelse(nchar(df[[name_col]]) > n,
                  paste0(substr(df[[name_col]], 1, n), "\u2026"), df[[name_col]])
  idx <- stats::ave(seq_along(short), short, FUN = seq_along)
  short[idx > 1] <- paste0(short[idx > 1], "#", idx[idx > 1])
  df$label <- short; df
}
bh_global <- function(p, m) {
  o <- order(p); ro <- order(o)
  adj <- p[o] * m / seq_along(p[o]); adj <- rev(cummin(rev(adj)))
  pmin(adj[ro], 1)
}

# Log2 values with IQR outliers flagged NA — used for scatter x-axis only.
# Mirrors the fence from zscore_trim (Q1 - 0.75*IQR / Q3 + 0.75*IQR) but
# skips imputation and z-scoring so the axis stays in interpretable log2 units.
.log2_trim_only <- function(x) {
  logged  <- log2(x + 1)
  orig_na <- is.na(logged)
  obs     <- logged[!orig_na]
  if (length(obs) < 2) return(rep(NA_real_, length(x)))
  qs  <- quantile(obs, c(0.25, 0.75), names = FALSE)
  iqr <- qs[2] - qs[1]
  v   <- logged
  v[!orig_na & (logged < qs[1] - 0.75 * iqr | logged > qs[2] + 0.75 * iqr)] <- NA
  v
}
metab_log_trim <- t(apply(metab_mat, 1, .log2_trim_only))

DFBETAS influence gate

Code 
# stable = no single sample moves the metabolite coefficient by more than
# 2/sqrt(n) of its SE. max_dfb = worst |DFBETAS| across samples (carried column).
dfbetas_group <- function(cid, ct, genes) {
  idx  <- which(row_ct == ct)
  mv   <- metab_tx[cid, match(metab_col[idx], colnames(metab_tx))]
  keep <- !is.na(mv); rows_use <- idx[keep]
  md   <- pb$metadata[rn[rows_use], ]
  lib  <- rowSums(pb$counts[rows_use, , drop = FALSE])
  base <- tibble(metab_trim = as.numeric(mv[keep]),
                 GW = md$GW, Sex = md$Sex, batch1_frac = md$batch1_frac)
  covs  <- c("GW", "Sex", "batch1_frac")
  keepc <- covs[vapply(covs, function(cv) length(unique(base[[cv]])) >= 2, logical(1))]
  fml   <- reformulate(c("metab_trim", keepc), response = "expr")

  purrr::map_dfr(genes, function(g) {
    if (!g %in% colnames(pb$counts))
      return(tibble(gene = g, max_dfb = NA_real_, dfb_cut = NA_real_, stable = NA))
    df <- base; df$expr <- log1p(pb$counts[rows_use, g] / lib * 1e6)
    n  <- nrow(df)
    if (length(unique(df$metab_trim)) < 2 || n < length(keepc) + 3)
      return(tibble(gene = g, max_dfb = NA_real_, dfb_cut = NA_real_, stable = NA))
    fit    <- lm(fml, data = df)
    dfb    <- stats::dfbetas(fit)[, "metab_trim"]      # SE-scaled single-deletion change
    cutoff <- 2 / sqrt(n)                              # standard DFBETAS cutoff
    tibble(gene = g, max_dfb = max(abs(dfb)), dfb_cut = cutoff,
           stable = max(abs(dfb)) <= cutoff)
  })
}

reporter <- hits |> group_by(Compound.ID, cell_type) |>
  group_modify(~ dfbetas_group(.y$Compound.ID, .y$cell_type, .x$gene)) |> ungroup()
hits <- hits |> left_join(reporter, by = c("Compound.ID", "cell_type", "gene"))

stable_hits <- hits |> filter(stable %in% TRUE)          # FILTER, not flag

Top-20 robust scatters

Five-panel scatter: raw log2 (all samples) | outlier-trimmed log2 | covariate-adjusted log2 || outlier-trimmed z-score | covariate-adjusted z-score. Log2 panels use free x-axis per metabolite; z-score panels share a fixed axis across all plots for cross-metabolite comparison. One pooled regression line per panel (the model includes Sex as a covariate, not a grouping variable); points colored by GW and shaped by Sex.

Code 
make_pair_plot <- function(cid, gene, ct, meta_name, z_x_lim) {
  idx      <- which(row_ct == ct)
  pos      <- match(metab_col[idx], colnames(metab_tx))
  present  <- !is.na(pos) & gene %in% colnames(pb$counts)
  rows_use <- idx[present]; pos_p <- pos[present]
  if (length(rows_use) < 3) return(NULL)

  expr <- as.numeric(log1p(edgeR::cpm(t(pb$counts[rows_use, , drop = FALSE]), log = FALSE)[gene, ]))
  md   <- pb$metadata[rn[rows_use], ]

  df <- tibble(
    metab_raw     = as.numeric(metab_log[cid, pos_p]),       # raw log2 (NAs for missing)
    metab_logtrim = as.numeric(metab_log_trim[cid, pos_p]),  # log2, outliers = NA
    metab_trim    = as.numeric(metab_tx[cid, pos_p]),        # z-score (model predictor)
    expr          = expr,
    Sex           = md$Sex,
    GW            = md$GW,
    batch1_frac   = md$batch1_frac
  )

  covs <- c("GW", "Sex", "batch1_frac")

  # ---- log2 panels (free x-axis per metabolite) ----
  # Compute x-limits from raw values (widest range, includes outliers)
  x_range <- range(df$metab_raw, na.rm = TRUE)
  x_pad   <- diff(x_range) * 0.05
  x_lim   <- c(x_range[1] - x_pad, x_range[2] + x_pad)
  x_lab   <- paste0(meta_name, " [log2 peak area]")

  # Shared layer list: shape in geom_point only → single regression line
  bl_log <- list(
    geom_point(aes(shape = Sex), size = 2.5, alpha = 0.85),
    geom_smooth(method = "lm", formula = y ~ x, se = TRUE, linewidth = 0.8),
    scale_color_viridis_c(name = "GW"),
    scale_shape_manual(values = c("M" = 16, "F" = 17)),
    coord_cartesian(xlim = x_lim),
    theme_minimal(base_size = 13)
  )

  # Panel 1: raw log2 — all observed samples, before any trimming
  df_raw <- df |> filter(!is.na(metab_raw))
  p_raw <- ggplot(df_raw, aes(metab_raw, expr, color = GW)) + bl_log +
    labs(title = "Raw log2 (all samples)", x = x_lab, y = "log1p(CPM)")

  # Panel 2: outlier-trimmed log2
  dft <- df |> filter(!is.na(metab_logtrim))
  keepc <- covs[vapply(covs, function(cv) length(unique(dft[[cv]])) >= 2, logical(1))]
  p_mid <- ggplot(dft, aes(metab_logtrim, expr, color = GW)) + bl_log +
    labs(title = "Outlier-trimmed (log2)", x = x_lab, y = "log1p(CPM)")

  # Panel 3: covariate-adjusted log2 (partial regression)
  fit_log  <- lm(reformulate(c("metab_logtrim", keepc), response = "expr"), data = dft)
  dft$part <- resid(fit_log) + coef(fit_log)[["metab_logtrim"]] * dft$metab_logtrim
  p_adj <- ggplot(dft, aes(metab_logtrim, part, color = GW)) + bl_log +
    labs(title = "Covariate-adjusted (log2)", x = x_lab, y = "log1p(CPM), adjusted")

  # ---- z-score panels (fixed x-axis across all plots) ----
  z_lab <- paste0(meta_name, " [z-score]")

  bl_z <- list(
    geom_point(aes(shape = Sex), size = 2.5, alpha = 0.85),
    geom_smooth(method = "lm", formula = y ~ x, se = TRUE, linewidth = 0.8),
    scale_color_viridis_c(name = "GW"),
    scale_shape_manual(values = c("M" = 16, "F" = 17)),
    coord_cartesian(xlim = z_x_lim),
    theme_minimal(base_size = 13)
  )

  # Panel 4: outlier-trimmed z-score (model predictor scale)
  dft_z   <- df |> filter(!is.na(metab_trim))
  keepc_z <- covs[vapply(covs, function(cv) length(unique(dft_z[[cv]])) >= 2, logical(1))]
  p_zmid  <- ggplot(dft_z, aes(metab_trim, expr, color = GW)) + bl_z +
    labs(title = "Outlier-trimmed (z-score)", x = z_lab, y = "log1p(CPM)")

  # Panel 5: covariate-adjusted z-score
  fit_z        <- lm(reformulate(c("metab_trim", keepc_z), response = "expr"), data = dft_z)
  dft_z$part_z <- resid(fit_z) + coef(fit_z)[["metab_trim"]] * dft_z$metab_trim
  p_zadj <- ggplot(dft_z, aes(metab_trim, part_z, color = GW)) + bl_z +
    labs(title = "Covariate-adjusted (z-score)", x = z_lab, y = "log1p(CPM), adjusted")

  # ---- 2-row layout: log2 row / z-score row ----
  (p_raw | p_mid | p_adj) / (plot_spacer() | p_zmid | p_zadj) +
    plot_annotation(title = paste0(gene, "  ~  ", meta_name, "   (", ct, ")")) +
    plot_layout(guides = "collect")
}
