Overview

Pathway analysis (ORA + MSEA) on DFBETAS-stable gene–metabolite associations from metabolite_gene_stability_dfbetas.qmd. Hits are pre-filtered to global FDR < 0.10, |logFC| ≥ 0.25, and no single sample moving the metabolite coefficient by more than 2/√n of its SE (|DFBETAS| ≤ 2/√n).

Three analyses:

  1. All cell types (union) — metabolites significant in ≥1 cell type
  2. Cell-type-specific — ORA + MSEA run independently per cell type
  3. Glia vs. neurons — metabolites associating primarily with glial or neuronal cell types, plus their exclusive pathway enrichments and shared pathway comparison

ORA: Fisher’s exact test; foreground = metabolites with ≥1 significant gene association (within FDR < 0.10, |logFC| ≥ 0.25); background = all measured metabolites.

MSEA: fgsea ranked by −log10(min raw P-value per metabolite) computed from the full parquet archive. This ranking is unsigned — it captures association strength without imposing a direction, since one metabolite can drive many gene associations with mixed signs.

  }

# ---- ORA (Fisher's exact, one-sided greater) ---------------------------------
run_ora <- function(foreground, background, pathway_list,
                    pathway_source_lookup, analysis_label) {
  cat("\n===", analysis_label, "===\n")
  cat("Foreground:", length(foreground), "| Background:", length(background), "\n")

  if (length(foreground) < 1) { cat("Empty foreground — skipping.\n"); return(NULL) }

  bg_only <- background[!background %in% foreground]

  results <- map_dfr(names(pathway_list), function(pw) {
    members <- pathway_list[[pw]]
    a <- sum(foreground %in% members)
    b <- length(foreground) - a
    c <- sum(bg_only %in% members)
    d <- length(bg_only) - c
    ft <- fisher.test(matrix(c(a, b, c, d), nrow = 2), alternative = "greater")
    tibble(
      pathway    = pw,
      n_hits     = a,
      set_size   = a + c,
      odds_ratio = as.numeric(ft$estimate),
      pval       = ft$p.value,
      hit_names  = paste(foreground[foreground %in% members], collapse = ";")
    )
  }) |>
    filter(n_hits > 0) |>
    mutate(padj = p.adjust(pval, method = "BH")) |>
    left_join(pathway_source_lookup, by = c("pathway" = "pathway_label_src")) |>
    arrange(padj, pval)

  cat("Pathways with \u22651 hit:", nrow(results),
      "| p < 0.05:", sum(results$pval < 0.05, na.rm = TRUE),
      "| FDR < 0.05:", sum(results$padj < 0.05, na.rm = TRUE), "\n")
  results
}

# ---- MSEA (fgsea on −log10 min P per metabolite) ----------------------------
# ranking_df : data frame with columns Name, min_P (from parquet group_by)
# background_names : all measured metabolite Names; those absent from ranking_df
#                    are assigned score = 0 (i.e. min_P = 1)
run_msea_metab <- function(ranking_df, background_names, pathway_list,
                           analysis_label, nperm = 10000, seed = 42) {
  cat("\n===", analysis_label, "===\n")

  ranked <- tibble(Name = background_names) |>
    left_join(ranking_df |> dplyr::select(Name, min_P), by = "Name") |>
    mutate(score = -log10(pmax(replace_na(min_P, 1), 1e-300))) |>
    arrange(desc(score)) |>
    dplyr::select(Name, score) |>
    deframe()

  cat("Metabolites in ranked list:", length(ranked), "\n")

  pathway_list_filt <- lapply(pathway_list, function(x) intersect(x, names(ranked)))
  pathway_list_filt <- pathway_list_filt[lengths(pathway_list_filt) >= 5]
  cat("Pathway sets after intersection:", length(pathway_list_filt), "\n")

  if (length(pathway_list_filt) == 0) {
    cat("No pathway sets with sufficient coverage — skipping.\n")
    return(NULL)
  }

  set.seed(seed)
  results <- fgsea(
    pathways    = pathway_list_filt,
    stats       = ranked,
    nPermSimple = nperm,
    minSize     = 5,
    maxSize     = 500,
    scoreType   = "pos"
  ) |>
    as_tibble() |>
    arrange(padj, pval) |>
    mutate(analysis = analysis_label)

  cat("Pathways tested:", nrow(results),
      "| p < 0.05:", sum(results$pval < 0.05, na.rm = TRUE),
      "| FDR < 0.05:", sum(results$padj < 0.05, na.rm = TRUE), "\n")
  results
}

plot_ora_dotplot <- function(results, analysis_label, top_n = 20) {
  if (is.null(results) || nrow(results) == 0) return(NULL)

  plot_data <- results |>
    filter(pval < 0.05) |>
    head(top_n) |>
    mutate(
      stars = case_when(padj < 0.01 ~ "**", padj < 0.05 ~ "*", TRUE ~ ""),
      pathway_base  = str_remove(pathway,
        " \\(KEGG\\)$| \\(Reactome\\)$| \\(WikiPathways\\)$| \\(PFOCR\\)$"),
      pathway_short = str_trunc(pathway_base, 45),
      pathway_label = paste0(pathway_short, " (", pathway_source, ")",
                             if_else(stars != "", paste0(" ", stars), ""))
    )

  if (nrow(plot_data) == 0) { message("No nominal hits for: ", analysis_label); return(NULL) }

  ggplot(plot_data,
         aes(x = log2(odds_ratio),
             y = reorder(pathway_label, log2(odds_ratio)),
             color = pathway_source, shape = pathway_source, size = set_size)) +
    geom_point(alpha = 0.85) +
    geom_vline(xintercept = 0, linetype = "dashed", color = "grey50") +
    scale_color_manual(name = "Database",
      values = c(KEGG = "#E41A1C", Reactome = "#377EB8",
                 WikiPathways = "#4DAF4A", PFOCR = "#FF7F00")) +
    scale_shape_manual(name = "Database",
      values = c(KEGG = 16, Reactome = 17, WikiPathways = 15, PFOCR = 18)) +
    scale_size_continuous(name = "Set size", range = c(2, 8)) +
    theme_minimal(base_size = 11) +
    labs(x = "log2(Odds Ratio)", y = NULL,
         title    = paste0("ORA \u2014 ", analysis_label),
         subtitle = paste0("Top ", nrow(plot_data), " by p-value (p < 0.05 shown)\n",
                           "* FDR < 0.05   ** FDR < 0.01")) +
    theme(plot.title    = element_text(hjust = 0.5),
          plot.subtitle = element_text(hjust = 0.5, color = "grey40", size = 9),
          axis.text.y   = element_text(size = 8),
          legend.position = "right")
}

display_results_table <- function(results, caption, top_n = 20,
                                  type = c("ora", "msea")) {
  type <- match.arg(type)
  if (is.null(results) || nrow(results) == 0) { cat("No results.\n"); return(invisible(NULL)) }

  if (type == "ora") {
    results |>
      head(top_n) |>
      mutate(odds_ratio = round(odds_ratio, 2),
             pval = signif(pval, 3), padj = signif(padj, 3),
             hit_names = str_trunc(hit_names, 60)) |>
      dplyr::select(pathway, pathway_source, n_hits, set_size,
                    odds_ratio, pval, padj, hit_names) |>
      rename(Pathway = pathway, Source = pathway_source,
             Hits = n_hits, `Set size` = set_size,
             OR = odds_ratio, `p-value` = pval, FDR = padj,
             `Hit metabolites` = hit_names) |>
      kable(caption = caption) |>
      kable_styling(bootstrap_options = c("striped", "condensed"),
                    full_width = FALSE, font_size = 11) |>
      print()
  } else {
    results |>
      left_join(pathway_source_lookup, by = c("pathway" = "pathway_label_src")) |>
      head(top_n) |>
      mutate(pval = signif(pval, 3), padj = signif(padj, 3),
             NES  = round(NES, 3),
             leadingEdge = map_chr(leadingEdge, ~ paste(head(.x, 5), collapse = ", "))) |>
      dplyr::select(pathway, pathway_source, pval, padj, NES, size, leadingEdge) |>
      rename(Pathway = pathway, Source = pathway_source,
             `p-value` = pval, FDR = padj, NES = NES,
             `Set size` = size, `Leading edge (top 5)` = leadingEdge) |>
      kable(caption = caption) |>
      kable_styling(bootstrap_options = c("striped", "condensed"),
                    full_width = FALSE, font_size = 11) |>
      print()
  }
}

    df |> mutate(leadingEdge = map_chr(leadingEdge, ~ paste(.x, collapse = ";")))
  else df
